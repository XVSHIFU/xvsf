{{- /* Suppress only an exact duplicate leading H1, preserving its anchor. */ -}} {{- $html := .Content -}} {{- $leading := findRE `(?s)^\s*]*>.*?` $html 1 -}} {{- with $leading -}} {{- $heading := index . 0 -}} {{- if eq (strings.TrimSpace ($heading | plainify | htmlUnescape)) (strings.TrimSpace ($.Title | plainify | htmlUnescape)) -}} {{- $anchor := replaceRE `(?s)]*)>.*?` `` $heading -}} {{- $html = replace $html $heading $anchor 1 -}} {{- end -}} {{- end -}} {{- return ($html | safeHTML) -}}
